# Can be used standalone or as native/modules2/guttur in Wirescape.
if(NOT TARGET wirescape_sdk2)
  cmake_minimum_required(VERSION 3.16)
  project(guttur_core VERSION 1.0 LANGUAGES CXX)
  include(CTest)
endif()
add_library(guttur_core STATIC src/engine.cpp src/tables.cpp src/tube.cpp src/room.cpp src/presets.cpp)
target_include_directories(guttur_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(guttur_core PUBLIC cxx_std_17)
set_target_properties(guttur_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(guttur_core PRIVATE /W4)
else()
  target_compile_options(guttur_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TARGET wirescape_sdk2)
 add_library(ws2_guttur SHARED module.cpp)
 target_include_directories(ws2_guttur PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 target_link_libraries(ws2_guttur PRIVATE wirescape_sdk2 guttur_core)
 # The app catalogue instantiates ws::Module<guttur::abi2::Module> in another
 # translation unit, so that target also needs the GUTTUR core symbols.
 if(TARGET wirescape_catalog_dependencies)
  target_link_libraries(wirescape_catalog_dependencies INTERFACE guttur_core)
 endif()
 add_executable(abi2_guttur_tests tests/wirescape_tests.cpp ../../tests2/allocation_watch.cpp)
 target_include_directories(abi2_guttur_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 target_link_libraries(abi2_guttur_tests PRIVATE wirescape_engine2 guttur_core)
 add_test(NAME abi2_guttur_tests COMMAND abi2_guttur_tests)
else()
add_executable(guttur_offline apps/offline_render.cpp)
target_link_libraries(guttur_offline PRIVATE guttur_core)
add_executable(guttur_benchmark apps/benchmark.cpp)
target_link_libraries(guttur_benchmark PRIVATE guttur_core)
if(BUILD_TESTING)
 add_executable(guttur_tests tests/engine_tests.cpp)
 target_link_libraries(guttur_tests PRIVATE guttur_core)
 add_test(NAME guttur_tests COMMAND guttur_tests)
endif()

endif() # Wirescape module or standalone executable/test targets
